import { FormEvent } from 'react';
import { Button, InlineMessage, TextAreaField } from '../design-system/components';
import type { VideoGenerationOption } from './video-generation.types';
 
type Props = { models: VideoGenerationOption[]; featureEnabled?: boolean; loading: boolean; error: string; onRetry: () => void; modelKey: string; setModelKey: (value: string) => void; prompt: string; setPrompt: (value: string) => void; aspectRatio: string; setAspectRatio: (value: string) => void; duration: string; setDuration: (value: string) => void; quality: string; setQuality: (value: string) => void; submitting: boolean; onSubmit: () => void };
export default function VideoGenerationForm(props: Props) {
  const model = props.models.find((item) => item.internalKey === props.modelKey); const promptError = !props.prompt.trim() ? 'توضیحات ویدیو را وارد کنید.' : props.prompt.trim().length < 3 ? 'توضیحات ویدیو باید حداقل ۳ کاراکتر باشد.' : model?.maxPromptLength != null && props.prompt.length > model.maxPromptLength ? `حداکثر ${model.maxPromptLength} کاراکتر مجاز است.` : '';
  const submit = (event: FormEvent) => { event.preventDefault(); props.onSubmit(); };
  return <form className="video-generation-form" onSubmit={submit} noValidate aria-busy={props.submitting}>
    <div className="video-section-heading"><h2>ساخت ویدیو از متن</h2><span className="video-coming-soon">تصویر به ویدیو به‌زودی</span></div>
    {props.loading ? <p className="video-loading" role="status">در حال دریافت مدل‌ها…</p> : props.error ? <div><p className="video-error" role="alert">{props.error}</p><Button type="button" variant="secondary" onClick={props.onRetry}>دریافت دوباره</Button></div> : props.featureEnabled === false ? <InlineMessage variant="help" text="ساخت ویدیو فعلاً توسط مدیر سامانه فعال نشده است." /> : !props.models.length ? <InlineMessage variant="help" text="فعلاً مدل فعالی برای ساخت ویدیو تنظیم نشده است." /> : <>
      <label className="ds-field" htmlFor="video-model"><span className="ds-field__label">مدل ساخت ویدیو</span><select id="video-model" className="ds-field__input" value={props.modelKey} onChange={(event) => props.setModelKey(event.target.value)}><option value="">مدل را انتخاب کنید</option>{props.models.map((item) => <option key={item.internalKey} value={item.internalKey}>{item.displayNameFa}</option>)}</select>{model?.descriptionFa ? <span className="ds-field__helper">{model.descriptionFa}</span> : null}</label>
      <TextAreaField id="video-prompt" label="توضیحات ویدیو" value={props.prompt} onChange={(event) => props.setPrompt(event.target.value)} placeholder="مثلاً یک سفر کوتاه در جنگل مه‌آلود، با حرکت آرام دوربین…" rows={5} maxLength={model?.maxPromptLength ?? undefined} errorText={promptError || undefined} helperText={model ? model.maxPromptLength == null ? 'حداکثر طولی از طرف مدل اعلام نشده است.' : `حداکثر ${model.maxPromptLength} کاراکتر` : 'ابتدا مدل را انتخاب کنید.'} />
      <div className="video-form-grid"><label className="ds-field" htmlFor="video-aspect-ratio"><span className="ds-field__label">نسبت تصویر</span><select id="video-aspect-ratio" className="ds-field__input" value={props.aspectRatio} onChange={(event) => props.setAspectRatio(event.target.value)} disabled={!model}>{model?.allowedAspectRatios.map((value) => <option key={value} value={value}>{value}</option>)}</select></label><label className="ds-field" htmlFor="video-duration"><span className="ds-field__label">مدت زمان</span><select id="video-duration" className="ds-field__input" value={props.duration} onChange={(event) => props.setDuration(event.target.value)} disabled={!model}>{model?.allowedDurations.map((value) => <option key={value} value={value}>{value}</option>)}</select></label>{model?.allowedQualities.length ? <label className="ds-field" htmlFor="video-quality"><span className="ds-field__label">کیفیت</span><select id="video-quality" className="ds-field__input" value={props.quality} onChange={(event) => props.setQuality(event.target.value)} disabled={!model}>{model.allowedQualities.map((value) => <option key={value} value={value}>{value}</option>)}</select></label> : null}</div>
      {model ? <p className="video-generation-form__cost">هزینه این مدل: {model.quotaUnits} واحد سهمیه. مانده سهمیه در پاسخ Backend فعلی موجود نیست.</p> : null}
      <Button type="submit" className="video-generation-form__submit" disabled={!model || Boolean(promptError) || props.submitting} loading={props.submitting}>{props.submitting ? 'در حال ثبت درخواست…' : 'ساخت ویدیو'}</Button>
    </>}
  </form>;
}
